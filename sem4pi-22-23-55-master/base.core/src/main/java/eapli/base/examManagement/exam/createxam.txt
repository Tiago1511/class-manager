grammar createExam;

exam_name: STRING+ NEWLINE*;

code_name: STRING+ NEWLINE*;

total_marks: NUMBER NEWLINE*;

passing_marks: NUMBER NEWLINE*;

duration_: NUMBER NEWLINE* STRING+ NEWLINE*;

text: STRING NEWLINE*
| text STRING+ NEWLINE*;

section: SECTION section_name MARKS section_marks QUESTIONS question_block;


section_name: text;

section_marks: NUMBER NEWLINE*;

question_block: question+ NEWLINE*;

question:  text answer_block ;

answer_block : answer+;


answer :text
|text NUMBER NEWLINE*;
